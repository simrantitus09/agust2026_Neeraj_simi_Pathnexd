Ansible, Terraform and Kubernetes Basics
Ansible -- Install Apache and Start Services

-name: Install and Start Apache
 host: all
 become: yes

 tasks:
   - name: Install Apache
     yum:
       name: httpd
       state: present
  
   - name: Start Apache Services
     service:
       name: httpd  
       state: started
       enabled: yes




Terraform — Create an EC2 Instance (t3.medium)

provider "aws" {
    region = "us-east-1"
}

resourse "aws_instance" "PathnexEC2" {
    ami = "ami-0abcd1234abcd1234"
    instance_type = "t3.medium"

    tags = {
        name = "Pathnex-T3"
    }
}

Kubernetes — Create a ReplicaSet with 3 Replicas

apiVersion: app/v1
kind: ReplicaSet

metadata:
  name: pathnex-replicaset

seec:
  replicas: 3

  selector:
    matchLabels:
      app: pathnex-app

  template:
    metadata:
      labels:
        app: pathnex-app

    spec:
      containers:
        - name: app
          image: nginx
          ports:
            - containerPort: 80
