#Jenkinsfile — With Git Integration and Testing

pipeline {
    aget any
    stages {
        stage('Clone Repository') {
            steps {
                git  'https://github.com/pathnex/sample-repo.git'
            }
        }
        stage('Build) {
            steps {
                echo 'Build project from Git repository'
            }
        }
        stage('Test') {
            steps {
                echo 'Running unit tests'

            }
        }
        stage('Deploy') {
            steps {
                echo 'Deploying application'
            }
        }
    } 
}

 #Ansible — Create User and Set Permissions

- name: Creat User and SEt Permissions
  host: all
  become: yes
  task:
    - name: Creat user
      user:
        name: pathnex
        state: present
        shell: /bin/bash
    - name: set permission for the user
      file:
        path: /home/pathnex
        owner: pathnex
        group: pathnex
        mode: '0755'
