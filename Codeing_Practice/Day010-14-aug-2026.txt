# Docker Integration with Jenkins & GitLab

# Ansible — Setup Docker Container for Nginx

- name: Setup Docker Container for Nginx
  host: all
  become: yes
  tasks:
    - name: Pull Nginx image
      docker_image:
        name: nginx
        source: pull
    - name: Run Nginx Container
      docker_container:
        name: nginx-Container
        image: nginx
        state: started
        publishd_ports:
          - "8080:80"
